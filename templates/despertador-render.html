/** * Cloudflare Worker - Intermediario "despertador" para Render * * Función: recibe la petición del usuario. Si el servicio en Render está dormido, * muestra una página de carga personalizada (sin que el usuario vea el interstitial * de Render) y lo despierta en segundo plano. Si ya está despierto, redirige directo. */ // ============================================================ // CONFIGURACIÓN - Cambia solo estas líneas si migras de servicio // ============================================================ const TARGET_URL = "https://gestion-transferencias-bancarias-2.onrender.com/landing"; const HEALTH_PATH = "/health"; // Tu endpoint real de health check // ============================================================ export default { async fetch(request) { const url = new URL(request.url); const targetUrl = TARGET_URL + url.pathname + url.search; // 1. Verificar si el servicio de Render ya está despierto const isAwake = await checkAwake(); if (isAwake) { // 2a. Ya está despierto: reenviar la petición real y devolver la respuesta return fetch(targetUrl, { method: request.method, headers: request.headers, body: request.body, redirect: "follow" }); } // 2b. Está dormido: devolver inmediatamente la página de carga personalizada // y lanzar el despertar en segundo plano (no bloquea la respuesta) wakeUpInBackground(); return new Response(loadingPage(request.url), { status: 200, headers: { "Content-Type": "text/html; charset=utf-8" } }); } }; /** * Comprueba si el servicio de Render responde en menos de 5 segundos. * Usa GET (en lugar de HEAD) porque FastAPI no registra HEAD por defecto. */ async function checkAwake() { try { const controller = new AbortController(); const timeout = setTimeout(() => controller.abort(), 5000); const res = await fetch(TARGET_URL + HEALTH_PATH, { method: "GET", signal: controller.signal }); clearTimeout(timeout); return res.ok || res.status < 500; } catch (e) { return false; } } /** * Despierta el servicio en segundo plano sin bloquear la respuesta al usuario. */ function wakeUpInBackground() { fetch(TARGET_URL + HEALTH_PATH, { method: "GET" }).catch(() => {}); } /** * Página HTML de carga personalizada. * Muestra un mensaje al usuario, y con JavaScript sondea el servicio hasta que esté listo. */ function loadingPage(originalUrl) { return ` Iniciando servicio...

Iniciando servicio

Estamos preparando todo para ti. Esto puede tardar unos segundos la primera vez del día.

Conectando...
No se pudo iniciar el servicio. Por favor, recarga la página.
`; }
